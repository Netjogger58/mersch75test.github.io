            let currentLang = 'lb';
            let currentName = '';
            let currentGender = 'm';

            function changeLang(lang) {
                currentLang = lang;
                document.querySelectorAll('#mersch75-dossier .lang-btn').forEach(btn => {
                    if (!btn.id.startsWith('gender')) {
                        btn.classList.remove('active');
                        if (btn.textContent.toLowerCase().includes(lang)) btn.classList.add('active');
                    }
                });
                document.querySelectorAll('#mersch75-dossier [data-i18n]').forEach(el => {
                    const key = el.getAttribute('data-i18n');
                    if (i18n[lang] && i18n[lang][key]) {
                        el.innerHTML = i18n[lang][key];
                    }
                });
                updateSalutation();
            }

            function updateName(val) {
                currentName = val;
                updateSalutation();
            }

            function setGender(g) {
                currentGender = g;
                document.getElementById('genderM').classList.toggle('active', g === 'm');
                document.getElementById('genderF').classList.toggle('active', g === 'f');
                updateSalutation();
            }

            function getEifelerPrefix(word, nextWord) {
                if (!nextWord) return word;
                const firstChar = nextWord.trim().charAt(0).toLowerCase();
                const keepChars = ['n', 'd', 't', 'z', 'h', 'a', 'e', 'i', 'o', 'u', 'ä', 'ë', 'é'];

                if (firstChar === 'y') return word;

                if (keepChars.includes(firstChar)) {
                    return word;
                }
                return word.slice(0, -1);
            }

            function updateSalutation() {
                const dict = i18n[currentLang];
                if (!dict) return;

                let prefix = "";
                const nameToUse = currentName ? currentName : "Member";

                if (currentLang === 'lb') {
                    if (currentGender === 'm') {
                        prefix = getEifelerPrefix(dict.salutation_m, nameToUse) + " ";
                    } else {
                        prefix = dict.salutation_f + " ";
                    }
                } else {
                    prefix = (currentGender === 'm' ? dict.salutation_m : dict.salutation_f) + " ";
                }

                const element = document.getElementById('dynamicNameSalutation');
                if (element) {
                    element.textContent = prefix + nameToUse + ",";
                }
            }

            return {
                init: function() {
                    changeLang('lb');
                },
                changeLang: changeLang,
                updateName: updateName,
                setGender: setGender
            };
        })();
